GPS UART

# apt-get install gpsd gpsd-clients

# apt-get install ntp

-- change /etc/default/gpsd  to configure ports for GPS data and boot option
-- new values at : https://github.com/guycole/perky-janus/blob/main/gpsd.default

# nano /etc/default/gpsd  (then copy paste gpsd.default from the above link)

-- For GPS data and PPS, get BB-UART1-GPS-00A0.dts compile and update BBB env
# from https://github.com/guycole/perky-janus/blob/main/BB-UART1-GPS-00A0.dts
# and compile at /home/ubuntu/bb.org-overlays  
bb.org-overlays# nano src/arm/BB-UART1-GPS-00A0.dts   
                          (copy paste, change P9.24, P9.26 MUX_MODE6 ==> MUX_MODE0)       
bb.org-overlays# make src/arm/BB-UART1-GPS-00A0.dtbo

-- Then copy BB-UART1-GPS-00A0.dtbo to /lib/firmware/
# cp src/arm/BB-UART1-GPS-00A0.dtbo /lib/firmware/

-- Update /boot/uEnv.txt to reference the new overlay
uboot_overlay_addr5=/lib/firmware/BB-UART1-GPS-00A0.dtbo

-- check folders
# ls  -al /lib/firmware/BB-UART1-GPS-00A0.dtbo
# ls  -a /sys/devices/platform/ocp/

PPS from gpsd  (Fixed on 18/1/2024) 
(updating gpsd config file at /etc/default/gpsd)

# nano /etc/default/gpsd

-- within gpsd config file
DEVICES="/dev/ttyO1 /dev/pps1"  ( disabled -- > DEVICES="/dev/ttyO1 /dev/pps0")

-- restart gpsd
# service gpsd restart        (**** to include 1PPS from /dev/pps1 *** )

--- test PPS 
# gpsmon
